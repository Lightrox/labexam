import javax.crypto.*; import javax.crypto.spec.*; import java.util.Scanner; public class AES { public static void main(String[] args) throws Exception { Scanner sc = new Scanner(System.in); System.out.print("Input your message: "); String msg = sc.nextLine(); KeyGenerator kg = KeyGenerator.getInstance("AES"); kg.init(128); SecretKey key = kg.generateKey(); SecretKeySpec spec = new SecretKeySpec(key.getEncoded(), "AES"); Cipher c = Cipher.getInstance("AES"); c.init(Cipher.ENCRYPT_MODE, spec); byte[] encrypted = c.doFinal(msg.getBytes()); String enc = new String(encrypted); System.out.println("Encrypted text: " + enc); c.init(Cipher.DECRYPT_MODE, spec); byte[] decrypted = c.doFinal(encrypted); System.out.println("Decrypted text: " + new String(decrypted)); } }

import java.math.*; import java.util.*; public class RSA { public static void main(String[] args) { Scanner sc = new Scanner(System.in); System.out.print("Enter a Prime number: "); BigInteger p = sc.nextBigInteger(); System.out.print("Enter another prime number: "); BigInteger q = sc.nextBigInteger(); BigInteger n = p.multiply(q); BigInteger phi = p.subtract(BigInteger.ONE).multiply(q.subtract(BigInteger.ONE)); BigInteger e; Random r = new Random(); do { e = BigInteger.valueOf(r.nextInt(phi.intValue() - 1)); } while (e.compareTo(BigInteger.TWO) <= 0 || !phi.gcd(e).equals(BigInteger.ONE)); BigInteger d = e.modInverse(phi); System.out.println("Encryption keys: " + e + ", " + n); System.out.println("Decryption keys: " + d + ", " + n); } }
